/* FLOATING HEARTS */

function createHeart(){

const heart =
document.createElement("div");

heart.className =
"floating-heart";

heart.innerHTML =
"♥";

heart.style.left =
Math.random()*100 + "vw";

heart.style.fontSize =
(12 + Math.random()*25) + "px";

heart.style.animationDuration =
(5 + Math.random()*5) + "s";

document.body.appendChild(heart);

setTimeout(
()=>heart.remove(),
10000
);

}

setInterval(createHeart,600);

</script>
